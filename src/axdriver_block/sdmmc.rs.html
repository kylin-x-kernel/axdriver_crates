1//! SD/MMC driver based on SDIO.
2
3use axdriver_base::{BaseDriverOps, DevError, DevResult, DeviceType};
4use simple_sdmmc::SdMmc;
5
6use crate::BlockDriverOps;
7
8/// A SD/MMC driver.
9pub struct SdMmcDriver(SdMmc);
10
11impl SdMmcDriver {
12    /// Creates a new [`SdMmcDriver`] from the given base address.
13    ///
14    /// # Safety
15    ///
16    /// The caller must ensure that `base` is a valid pointer to the SD/MMC controller's
17    /// register block and that no other code is concurrently accessing the same hardware.
18    pub unsafe fn new(base: usize) -> Self {
19        Self(SdMmc::new(base))
20    }
21}
22
23impl BaseDriverOps for SdMmcDriver {
24    fn device_type(&self) -> DeviceType {
25        DeviceType::Block
26    }
27
28    fn device_name(&self) -> &str {
29        "sdmmc"
30    }
31}
32
33impl BlockDriverOps for SdMmcDriver {
34    fn num_blocks(&self) -> u64 {
35        self.0.num_blocks()
36    }
37
38    fn block_size(&self) -> usize {
39        SdMmc::BLOCK_SIZE
40    }
41
42    fn read_block(&mut self, block_id: u64, buf: &mut [u8]) -> DevResult {
43        let (blocks, remainder) = buf.as_chunks_mut::<{ SdMmc::BLOCK_SIZE }>();
44
45        if !remainder.is_empty() {
46            return Err(DevError::InvalidParam);
47        }
48
49        for (i, block) in blocks.iter_mut().enumerate() {
50            self.0.read_block(block_id as u32 + i as u32, block);
51        }
52
53        Ok(())
54    }
55
56    fn write_block(&mut self, block_id: u64, buf: &[u8]) -> DevResult {
57        let (blocks, remainder) = buf.as_chunks::<{ SdMmc::BLOCK_SIZE }>();
58
59        if !remainder.is_empty() {
60            return Err(DevError::InvalidParam);
61        }
62
63        for (i, block) in blocks.iter().enumerate() {
64            self.0.write_block(block_id as u32 + i as u32, block);
65        }
66
67        Ok(())
68    }
69
70    fn flush(&mut self) -> DevResult {
71        Ok(())
72    }
73}
